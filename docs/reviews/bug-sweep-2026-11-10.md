## HIGH

### H1. `pipeline/tests-step-offline-run`: the Tests step runs the real collector and fails every CI run
- **Where:** `collector/tests/test_run.py:87-95`; `.github/workflows/daily-update.yml:115-116` (Tests step, no `env`);
  `collector/__main__.py:346-349`; `sitegen/build.py:208-215`; `sitegen/config.py:75`.
- **What:** `test_second_offline_run_changes_no_committed_file` is `skipif` the document store is incomplete. Locally
  that is always true, so it is skipped. In CI the store is restored from the Actions cache, so it runs
  `M.main(["run", "--offline"])` twice against the real repo paths. The Tests step sets neither
  `RECORDS_BUILD_TARGET` nor `RECORDS_SITE_URL`, so `DEFAULT_TARGET` falls back to `preview` (cap 500 files) and the
  run rebuilds `build/site` in place, then `write_publish_manifest` raises.
- **Proof:** replay of that call with the Tests step's environment (parse, finance and agenda steps stubbed, the site
  step real; test text in the appendix, `test_ci_offline_run_test_clobbers_site.py`):
  ```
  sitegen.build.BuildError: site has 505 files; the limit for the None target is 500
  collector/__main__.py:349: in cmd_run   write_publish_manifest(site)
  1 failed in 153s
  ```
- **Impact:** `pytest -q -x` fails, so no site artifact, no data commit, no deploy, on every scheduled run. If the file
  count were ever under 500 it would be worse: the step would pass after overwriting the checked `build/site` (pages
  target, absolute feed links, morning note) with a preview build, and that is what would be uploaded and deployed,
  after the site checks had already passed on the other build.
- **Fix (one line):** make that test use the `sandbox` paths (or mark it out of the CI run), and set the two build
  variables at job level so no step can rebuild as preview. **Owner:** devops-engineer + backend-engineer.

### H2. `pipeline/changes-jsonl-rejected`: the first routine document change breaks the push job, every day after

### H3. `city-seats/sitting-members-served-previously`: all 12 wards show no current member; 11 sitting members are under "Served previously"

### H4. `city-votes/minutes-date-unchecked`: minutes of another meeting are read as the listed meeting and "verified"

### H5. `topics/number-prefix-not-typed`: city ordinances and resolutions since 2022 are typed "other"; four topic feeds are silent for the City Council
- **Where:** `sitegen/views.py:1408-1413` (`item_types`); `config/site.yml` `item_types` (`ordinance` and `resolution`
  have `prefixes: []`, title words only); consumers `sitegen/feeds.py:160`, `sitegen/views.py:1441,1448` (search type
  filter, `site/static/search.js:107`).
- **What:** type comes from a number prefix or from words in the title. City items since 2022 carry a number
  ("ORDINANCE 09-2025-26") and an empty title, and "ORDINANCE"/"RESOLUTION" are not prefixes.
- **Proof:**

### test_ci_offline_run_test_clobbers_site.py
```python
"""pipeline/tests-step-rebuilds-site: in CI the document store is complete, so collector/tests/test_run.py::
test_second_offline_run_changes_no_committed_file is NOT skipped and runs the real `collector run --offline` twice, in
the workflow's "Tests" step, where RECORDS_BUILD_TARGET / RECORDS_SITE_URL are not set. This replays that call with the
parse, finance and agenda steps stubbed (as test_run.py's own `sandbox` fixture stubs them) and the site step real."""
import json
import os
import shutil
from pathlib import Path

import pytest

import collector.__main__ as M

ROOT = Path(M.__file__).resolve().parent.parent


def test_offline_run_in_the_tests_step(tmp_path, monkeypatch):
    assert not os.environ.get("RECORDS_BUILD_TARGET") and not os.environ.get("RECORDS_SITE_URL")   # the Tests step
    data = tmp_path / "data"
    data.mkdir()
    for name in ("facts.json", "finance.json"):
        shutil.copy(ROOT / "data" / name, data / name)
    facts = json.loads((data / "facts.json").read_text())
    report = {"gates": {"roll_calls": 1, "verified": 1, "verified_rate": 1.0, "held_back": 0,
                        "held_back_source_error": 0, "verified_rate_gate": {"min": 0.9, "pass": True},
                        "g4": {"status": "pass", "uniform_share": 0.0, "flagged": []}}, "pass": True}

    class B:
        unresolved = {}

    def fake_build(store, log, use_ocr=True):
        log["parsed"] = 1
        return facts, report, B()

    monkeypatch.setattr(M, "FACTS", data / "facts.json")
    monkeypatch.setattr(M, "FINANCE", data / "finance.json")
    monkeypatch.setattr(M, "BUILD", tmp_path / "build")          # stands for the repo's build/, which CI uploads
    monkeypatch.setattr(M, "cmd_build", fake_build)
    monkeypatch.setattr(M, "missing_inputs", lambda store: [])
    monkeypatch.setattr(M, "shrink_reason", lambda *a, **k: None)
    monkeypatch.setattr(M, "run_finance", lambda *a, **k: 0)
    monkeypatch.setattr(M, "run_agendas", lambda *a, **k: None)
    code = M.main(["run", "--offline"])
    site = tmp_path / "build" / "site"
    print("exit", code, "| morning notes:", (site / "morning").is_dir(), "| transcripts:",
          len(list(site.glob("*/transcripts/*.html"))), "| files:", sum(1 for p in site.rglob("*") if p.is_file()))
    assert (site / "morning" / "latest.json").is_file(), "build/site is no longer the pages-target build"
```

### test_archive_change_breaks_push.py
```python
"""pipeline/changes-jsonl-rejected: a routine change of the purchasing archive table writes <data>/changes.jsonl, and the
push job's "Validate refreshed data" step (only *.json|*.yml) then refuses the whole data artifact."""
import os
import shutil
import subprocess
from pathlib import Path

import yaml

from collector.tests.test_fin_fetch import ARCHIVE, FIXTURE_HTML, _fin_fetch

ROOT = Path(__file__).resolve()
WF = yaml.safe_load(Path("/tmp/bh/.github/workflows/daily-update.yml").read_text())


def test_archive_table_change_passes_the_push_validation(tmp_path):
    data = tmp_path / "data"            # stands for data/: manifest.json and (store.py) changes.jsonl live side by side
    data.mkdir()
    routes = {ARCHIVE: (200, FIXTURE_HTML, {"Content-Type": "text/html"})}
    _fin_fetch(data, routes)                                                   # day 1: first collection
    routes[ARCHIVE] = (200, FIXTURE_HTML.replace(b"MAY 4, 2026", b"MAY 5, 2026"), {"Content-Type": "text/html"})
    log, _, _, _ = _fin_fetch(data, routes)                                    # day 2: the county edits one row
    assert log["archive"]["status"] == "changed"
    shutil.rmtree(data / "store", ignore_errors=True)                          # the artifact excludes data/store
    for p in data.glob("*.pdf"):
        p.unlink()
    step = next(s for s in WF["jobs"]["push"]["steps"] if s.get("name") == "Validate refreshed data")
    art = tmp_path / "runner" / "records-data"
    shutil.copytree(data, art, ignore=shutil.ignore_patterns("*.pdf", "*.part"))
    r = subprocess.run(["bash", "-c", step["run"]], env={**os.environ, "RUNNER_TEMP": str(tmp_path / "runner")},
                       capture_output=True, text=True)
    print(sorted(p.name for p in art.iterdir()), r.stdout.strip().replace(str(tmp_path), "<tmp>"))
    assert r.returncode == 0, "push job refuses the refreshed data"
```

### test_wrong_minutes_file.py

```python
"""city-votes/minutes-date-unchecked: a minutes file that prints another meeting's date, posted once, is read as the
listed meeting and its roll calls are 'verified'."""
from collector import agenda_center as ac
from collector.tests.test_civicclerk_minutes import API, CAPS_2025, FULL_2025, built, cc_rows
from collector.tests.test_city_minutes import entry
from collector.city_minutes import called_to_order_on


def test_minutes_printing_another_date_are_not_read_as_the_listed_meeting():
    cfg = ac.load_config(ac.config_path("clarksville-tn"))
    assert called_to_order_on(CAPS_2025) == "2025-08-07"          # the file's own call-to-order date
    rows = cc_rows()
    rows[1] = {**rows[1], "date": "2025-09-04"}                   # the clerk attached the August minutes to September
    facts, summary = built(cfg, [FULL_2025, CAPS_2025], rows,
                           [entry(rows[0]["minutes_url"], "minutes", "2025-02-06", 1),
                            entry(rows[1]["minutes_url"], "minutes", "2025-09-04", 2)])
    sep = {m["date"]: m for m in facts["meetings"]}["2025-09-04"]
    rcs = [r for r in facts["roll_calls"] if r["meeting_id"] == sep["id"]]
    print("minutes_status:", sep["minutes_status"], "| roll calls:", [r["verification"]["status"] for r in rcs],
```
